MATH ENGINES (SMA, EMA, RSI, STOCH RSI) // ========================================== function calcMA(candles, type, period) { const res = []; const k = 2 / (period + 1); let emaPrev = null; for (let i = 0; i < candles.length; i++) { const close = candles[i].close; if (type === 'SMA') { if (i >= period - 1) { let sum = 0; for (let j = 0; j < period; j++) sum += candles[i - j].close; res.push({ time: candles[i].time, value: Number((sum / period).toFixed(2)) }); } } else { if (i === period - 1) { let sum = 0; for (let j = 0; j < period; j++) sum += candles[i - j].close; emaPrev = sum / period; res.push({ time: candles[i].time, value: Number(emaPrev.toFixed(2)) }); } else if (i >= period && emaPrev !== null) { emaPrev = (close - emaPrev) * k + emaPrev; res.push({ time: candles[i].time, value: Number(emaPrev.toFixed(2)) }); } } } return res; } function calcRSIArray(candles, period) { const rsiOut = new Array(candles.length).fill(null); if (candles.length <= period) return rsiOut; let gainSum = 0, lossSum = 0; for (let i = 1; i <= period; i++) { const diff = candles[i].close - candles[i - 1].close; if (diff >= 0) gainSum += diff; else lossSum -= diff; } let avgGain = gainSum / period; let avgLoss = lossSum / period; rsiOut[period] = avgLoss === 0 ? -diff : 0; avgGain = (avgGain * (period - 1) + g) / period; avgLoss = (avgLoss * (period - 1) + l) / period; rsiOut[i] = avgLoss === 0 ? 100 : 100 - (100 / (1 + avgGain / avgLoss)); } return rsiOut; } function calcStochRSI(candles, rsiLen, stochLen, smoothK, smoothD) { const rsiArr = calcRSIArray(candles, rsiLen); const rawStoch = new Array(candles.length).fill(null); for (let i = 0; i < candles.length; i++) { if (i < rsiLen + stochLen - 1) continue; let minR = Infinity, maxR = -Infinity; let valid = true; for (let j = 0; j < stochLen; j++) { const v = rsiArr[i - j]; if (v === null) { valid = false; break; } if (v < minR) minR = v; if (v > maxR) maxR = v; } if (valid && maxR !== minR) { rawStoch[i] = 100 * ((rsiArr[i] - minR) / (maxR - minR)); } } const kArr = new Array(candles.length).fill(null); for (let i = 0; i < candles.length; i++) { if (i < smoothK - 1) continue; let sum = 0, ok = true; for (let j = 0; j < smoothK; j++) { if (rawStoch[i - j] === null) { ok = false; break; } sum += rawStoch[i - j]; } if (ok) kArr[i] = sum / smoothK; } const kSeries = [], dSeries = []; for (let i = 0; i < candles.length; i++) { if (kArr[i] !== null) { kSeries.push({ time: candles[i].time, value: Number(kArr[i].toFixed(2)) }); } if (i >= smoothD - 1) { let sum = 0, ok = true; for (let j = 0; j < smoothD; j++) { if (kArr[i - j] === null) { ok = false; break; } sum += kArr[i - j]; } if (ok) { dSeries.push({ time: candles[i].time, value: Number((sum / smoothD).toFixed(2)) }); } } } return { kSeries, dSeries }; } // ========================================== // 3.
